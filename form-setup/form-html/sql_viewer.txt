-- 1. Main Treatment Outcomes (New, Relapse, Unknown, Other Retreatment categories)
SELECT 
    parent_ou.name AS "State",
    ou.name AS "LGA",
    prev_de.name AS "Notification",
    CASE 
        WHEN EXTRACT(MONTH FROM prev_p.startdate) = 1 THEN 'Jan-Mar ' || TO_CHAR(prev_p.startdate, 'YYYY')
        WHEN EXTRACT(MONTH FROM prev_p.startdate) = 4 THEN 'Apr-Jun ' || TO_CHAR(prev_p.startdate, 'YYYY')
        WHEN EXTRACT(MONTH FROM prev_p.startdate) = 7 THEN 'Jul-Sep ' || TO_CHAR(prev_p.startdate, 'YYYY')
        WHEN EXTRACT(MONTH FROM prev_p.startdate) = 10 THEN 'Oct-Dec ' || TO_CHAR(prev_p.startdate, 'YYYY')
        ELSE TO_CHAR(prev_p.startdate, 'Mon-YYYY')
    END AS "Notified Period",
    MAX(pn.total_notified) AS "Notified value",
    co.patient_category_row AS "Treatment Outcome",
    CASE 
        WHEN EXTRACT(MONTH FROM curr_p.startdate) = 1 THEN 'Jan-Mar ' || TO_CHAR(curr_p.startdate, 'YYYY')
        WHEN EXTRACT(MONTH FROM curr_p.startdate) = 4 THEN 'Apr-Jun ' || TO_CHAR(curr_p.startdate, 'YYYY')
        WHEN EXTRACT(MONTH FROM curr_p.startdate) = 7 THEN 'Jul-Sep ' || TO_CHAR(curr_p.startdate, 'YYYY')
        WHEN EXTRACT(MONTH FROM curr_p.startdate) = 10 THEN 'Oct-Dec ' || TO_CHAR(curr_p.startdate, 'YYYY')
        ELSE TO_CHAR(curr_p.startdate, 'Mon-YYYY')
    END AS "Reporting Period",
    SUM(co.outcome_value) AS "Outcome value",
    CASE 
        WHEN MAX(pn.total_notified) IS NULL THEN '⚠️ Data entry missing for the corresponding quarter last year; unable to verify consistency.'
        WHEN SUM(co.outcome_value) = MAX(pn.total_notified) THEN '✅ Values match between current quarter and same period last year.'
        ELSE '❌ Current quarter row total (' || SUM(co.outcome_value) || ') does not equal last year notification total (' || MAX(pn.total_notified) || ').'
    END AS "Discrepancy detected"
FROM current_outcomes co
JOIN organisationunit ou ON co.sourceid = ou.organisationunitid
LEFT JOIN organisationunit parent_ou ON ou.parentid = parent_ou.organisationunitid
JOIN period curr_p ON co.periodid = curr_p.periodid
JOIN period prev_p ON prev_p.periodtypeid = curr_p.periodtypeid
    AND prev_p.startdate = (curr_p.startdate - INTERVAL '1 year')
LEFT JOIN dataelement prev_de ON prev_de.uid = co.target_notified_de_uid
LEFT JOIN previous_notifications pn ON pn.sourceid = co.sourceid 
    AND pn.periodid = prev_p.periodid 
    AND pn.dataelementid = prev_de.dataelementid
GROUP BY 
    parent_ou.name,
    ou.name,
    co.patient_category_row,
    curr_p.startdate,
    prev_p.startdate,
    prev_de.name

    END AS "Notified Period",

    curr_de.name AS "Treatment Outcome",
    CASE 
        WHEN EXTRACT(MONTH FROM curr_p.startdate) = 1 THEN 'Jan-Mar ' || TO_CHAR(curr_p.startdate, 'YYYY')

    END AS "Reporting Period",

    END AS "Discrepancy detected"
FROM datavalue curr_dv
JOIN dataelement curr_de ON curr_dv.dataelementid = curr_de.dataelementid
JOIN organisationunit ou ON curr_dv.sourceid = ou.organisationunitid
LEFT JOIN organisationunit parent_ou ON ou.parentid = parent_ou.organisationunitid
JOIN period curr_p ON curr_dv.periodid = curr_p.periodid
JOIN period prev_p ON prev_p.periodtypeid = curr_p.periodtypeid
    AND prev_p.startdate = (curr_p.startdate - INTERVAL '1 year')
LEFT JOIN datavalue prev_dv ON prev_dv.sourceid = curr_dv.sourceid 
    AND prev_dv.periodid = prev_p.periodid
    AND prev_dv.dataelementid = (SELECT dataelementid FROM dataelement WHERE uid = 'DEcjC6PdjCf')
LEFT JOIN dataelement prev_de ON prev_dv.dataelementid = prev_de.dataelementid
WHERE curr_de.uid = 'yfJs1ON43xK'
  AND curr_dv.value IS NOT NULL
GROUP BY 
    parent_ou.name,
    ou.name,
    curr_de.name,
    curr_p.startdate,

    prev_de.name
